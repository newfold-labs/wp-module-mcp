34     * and unversioned (e.g., `wc-analytics`) namespaces.
35     *

37     *   Base: "wc"           → Returns: "wc/v3" (if v1, v2, v3 are registered)
38     *   Base: "wp"           → Returns: "wp/v2"

43     * @return string|null The latest versioned namespace, or null if not found.

46        $server     = rest_get_server();
47        $namespaces = $server->get_namespaces();
48
49        $base_namespace = trim( $base_namespace, '/' );
50        $versions       = array();

53            // Exact match (unversioned namespace like "wc-analytics")
54            if ( $ns === $base_namespace ) {

56            }

65            return null;
66        }
